⭐ Top voted on Reddit

Top N per group without CTEs

How do we get the top 3 performers in each department?

With CTE
WITH ranked AS (
    SELECT *,
        ROW_NUMBER() OVER (
            PARTITION BY dept
            ORDER BY salary DESC
        ) AS rn
    FROM employees
)
SELECT * FROM ranked
WHERE rn <= 3;
With QUALIFY
SELECT * FROM employees
QUALIFY ROW_NUMBER() OVER (
    PARTITION BY dept
    ORDER BY salary DESC
) <= 3;
  • Intent is explicity
  • No more nested subsequeries, redundant columns
  • Half of the code!!
Non-AI SQL
⭐ Top voted on Reddit

Stop repeating yourself

Why do I have to list every non-aggregated column twice?

Before
SELECT
    department, role, location,
    COUNT(*) AS headcount,
    AVG(salary) AS avg_salary
FROM employees
GROUP BY department, role, location;
With GROUP BY ALL
SELECT
    department, role, location,
    COUNT(*) AS headcount,
    AVG(salary) AS avg_salary
FROM employees
GROUP BY ALL;
  • Less Typing
  • Fewer bugs
  • No more 'not a group by expression' error!
Non-AI SQL

Pipe Operator (`->>`)

How do we stay in flow while chaining SQL steps?

Before 1
SHOW WAREHOUSES;

SELECT "name", "state", "type", "size"
  FROM TABLE(RESULT_SCAN(LAST_QUERY_ID(-1)));
Before 2 (traditional approach)
-- Pseudo code (~35+ lines): many steps + query_id tracking
BEGIN TRANSACTION;
INSERT INTO test_sql_pipe_dml VALUES (1, 2);
SET q1 = LAST_QUERY_ID();
DELETE FROM test_sql_pipe_dml WHERE a = 1;
SET q2 = LAST_QUERY_ID();
UPDATE test_sql_pipe_dml SET b = 2;
SET q3 = LAST_QUERY_ID();
COMMIT;
SELECT
  ( SELECT 1 FROM TABLE(RESULT_SCAN($q1)) ) AS "Inserted rows",
  ( SELECT 2 FROM TABLE(RESULT_SCAN($q2)) ) AS "Deleted rows",
  ( SELECT 3 FROM TABLE(RESULT_SCAN($q3)) ) AS "Updated rows";
After 1
-- This statement uses the pipe operator to produce the same results:
SHOW WAREHOUSES
  ->> SELECT "name", "state", "type", "size" FROM $1;
After 2
BEGIN TRANSACTION
  ->> INSERT INTO test_sql_pipe_dml VALUES (1, 2)
  ->> DELETE FROM test_sql_pipe_dml WHERE a = 1
  ->> UPDATE test_sql_pipe_dml SET b = 2
  ->> COMMIT
  ->> SELECT
        (SELECT $1 FROM $4) AS "Inserted rows",
        (SELECT $1 FROM $3) AS "Deleted rows",
        (SELECT $1 FROM $2) AS "Updated rows";

+---------------+--------------+--------------+
| Inserted rows | Deleted rows | Updated rows |
|---------------+--------------+--------------|
|             1 |            2 |            3 |
+---------------+--------------+--------------+

Stay in the flow!!

What's so great about the great lakes

  • Intent is explicit and linear
  • Fewer temporary variables/query IDs
  • Much easier debugging in sequence
Non-AI SQL

MAX_BY / MIN_BY

Who is on a 996 work pattern (9am to 9pm, 6 days/week)?

-- 996 = 9am to 9pm, 6 days/week (72 hrs)
-- Traditional pattern: aggregate first, then filter
WITH weekly_hours AS (
  SELECT employee_id, employee_name, department, week_start_date,
         SUM(hours_worked) AS total_hours
  FROM employee_timesheets
  GROUP BY employee_id, employee_name, department, week_start_date
)
SELECT employee_id, employee_name, department, week_start_date, total_hours
FROM weekly_hours
WHERE total_hours >= 72;

-- MAX_BY: get the heaviest week per employee + filter 996 pattern
WITH per_week AS (
  SELECT employee_id, employee_name, department, week_start_date,
         SUM(hours_worked) AS total_hours
  FROM employee_timesheets
  GROUP BY ALL
)
SELECT employee_id,
       MAX_BY(OBJECT_CONSTRUCT(
         'employee_name', employee_name,
         'department', department,
         'week_start_date', week_start_date,
         'hours', total_hours
       ), total_hours) AS peak_week
FROM per_week
GROUP BY employee_id
HAVING MAX(total_hours) >= 72;

What's so great about the great lakes

  • Intent is clear in one query
  • No extra ranking CTE stack
  • Compact and easier to maintain

996 means 72+ hours in a week. This pattern flags likely sustained overtime risk.

Non-AI SQL

MAX_BY for AI usage by department

Find employee with max token usage by department.

Before (traditional CTE)
WITH usage_rollup AS (
    SELECT department, employee_id, employee_name,
           SUM(tokens_used) AS total_tokens
    FROM employee_ai_usage
    GROUP BY department, employee_id, employee_name
),
ranked AS (
    SELECT *,
           ROW_NUMBER() OVER(
             PARTITION BY department
             ORDER BY total_tokens DESC
           ) AS rn
    FROM usage_rollup
)
SELECT department, employee_id, employee_name, total_tokens
FROM ranked
WHERE rn = 1;
After (intent clear + compact)
SELECT department, MAX_BY( object_construct(
            'employee_id', employee_id
          , 'employee_name', employee_name
          , 'total_tokens', SUM(tokens_used)
        ),
        SUM(tokens_used)) AS top_employee
FROM employee_ai_usage
GROUP BY ALL;

-- Example output:
-- Engineering | {"employee_id":"E101","employee_name":"Alice Chen","total_tokens":184920}
-- Marketing   | {"employee_id":"E214","employee_name":"Ben Wong","total_tokens":132400}
-- Finance     | {"employee_id":"E089","employee_name":"Nina Patel","total_tokens":118230}
Non-AI SQL

CONCAT_WS handles nulls gracefully

Build full names with optional middle names.

Before
SELECT
  TRIM(
    first_name || ' ' ||
    COALESCE(middle_name || ' ', '') ||
    last_name
  ) AS full_name
FROM employees;
After
SELECT
  CONCAT_WS(' ', first_name, middle_name, last_name) AS full_name
FROM employees;

-- Example output:
-- first_name | middle_name | last_name | full_name
-- Raj        | NULL        | Thakkar   | Raj Thakkar
-- Ada        | Grace       | Hopper    | Ada Grace Hopper

What's so great about the great lakes

  • Handles NULLs gracefully
  • Cleaner and more readable than nested COALESCE chains
  • Fewer string concatenation bugs

AI_EXTRACT (AKA Inbuilt OCR)

How do we process 50,000 invoice PDFs monthly?

Before
Build OCR pipeline + Python service
↓
Map fields manually
↓
Add retries + reprocessing queues
↓
Maintain multiple moving parts
After
SELECT AI_EXTRACT(
    file => TO_FILE('@DEMO_AUS_DB.BRONZE.call_recordings_stage', 'stug-2026-v5.pdf'),
    responseFormat => {'title': 'What is the title of the presentation?',
                       'pages': 'how many pages in the presentation?',

AI_REDACT (AKA Inbuilt heavy black marker)

How do we safely share logs with external vendors?

Before
Regex patterns for each PII type
↓
Custom masking rules per source
↓
High maintenance + misses edge cases
After
SELECT AI_REDACT(
    'Contact John Smith at john@email.com or 0412 345 678',
    ['NAME', 'EMAIL', 'PHONE_NUMBER']
) AS sanitized;

-- Example output:
-- Contact [NAME] at [EMAIL] or [PHONE_NUMBER]

Valid categories: NAME, EMAIL, PHONE_NUMBER, DATE_OF_BIRTH, GENDER, AGE, ADDRESS, NATIONAL_ID, PASSPORT, TAX_IDENTIFIER, PAYMENT_CARD_DATA, DRIVERS_LICENSE, IP_ADDRESS.
AI_REDACT uses a Snowflake managed LLM model [proprietary/undisclosed].

What's so great about the great lakes

  • PII protection directly in SQL workflows
  • No external masking service required
  • Easy to operationalise in shared datasets and logs

AI_COMPLETE (AKA Inbuilt ChatGPT)

How do we analyse a graph from a stage directly in SQL?

Before
Export image to separate app
↓
Prompt manually outside Snowflake
↓
Copy back insights to warehouse

What's so great about the great lakes

  • Awesome way to integrate AI in your data pipelines
  • Multimodal
  • Data stays within Snowflake purview
  • New models added regularly
  • Token based pricing

21+ models supported from [anthropic, OpenAI, Openweights].

After
SELECT AI_COMPLETE(
    'claude-sonnet-4-6',
    PROMPT(
        'Analyze this sales graph. Describe the trends, key insights, and any notable patterns.',
        TO_FILE('@DEMO_AUS_DB.BRONZE.YOUR_STAGE', 'sales_graph.png')
    )
) AS analysis;
-- Sample output: "Sales trend is steadily growing with a strong Q4 peak..."

SELECT AI_COMPLETE(
    'claude-sonnet-4-6',
    PROMPT(
        'Look at this sales graph and answer:
         1. What is the overall trend (growing, declining, flat)?
         2. Which period had the highest sales?
         3. Are there any seasonal patterns?
         4. What recommendations would you make based on this data?',
        TO_FILE('@DEMO_AUS_DB.BRONZE.YOUR_STAGE', 'sales_graph.png')
    )
) AS detailed_analysis;
-- Sample output: "1) Growing 2) Q4 3) Seasonal spikes around holidays 4) Improve inventory planning"

SELECT AI_COMPLETE(
    'claude-sonnet-4-6',
    PROMPT(
        'Analyze this sales chart and return a JSON with: trend, peak_period, low_period, growth_rate_estimate, key_observations',
        TO_FILE('@DEMO_AUS_DB.BRONZE.YOUR_STAGE', 'sales_graph.png')
    ),
    {'response_format': {'type': 'json'}}
) AS structured_analysis;
-- Sample output: {"trend":"growing","peak_period":"Q4","low_period":"Q1","growth_rate_estimate":"~11%","key_observations":["seasonality","strong close"]}

More SQL Features

Classic SQL "Dad Jokes" Generator

SELECT AI_COMPLETE('llama3-70b', 'Generate a classic SQL Dad Joke.');

The Bar Visit: A SQL query walks into a bar, walks up to two tables, and asks, "Can I JOIN you?"

01
CREATE OR ALTER
Idempotent DDL. Update or create in one command.
CREATE OR ALTER TABLE employee_dim (id NUMBER, name STRING);
02
EXCLUDE
Remove noisy columns quickly.
SELECT * EXCLUDE (raw_payload, pii_email) FROM employee_360;
03
DIV0
Safe division with zero handling.
SELECT DIV0(total_cost, total_revenue) AS ratio FROM kpi_daily;
04
SEQ1 / SEQ2 / SEQ4 / SEQ8 with GENERATOR
Generate millions of rows quickly for performance/load testing.
CREATE TABLE load_test AS SELECT SEQ8() AS id, UUID_STRING() AS guid, RANDOM() AS random_val FROM TABLE(GENERATOR(ROWCOUNT => 10000000));
05
SOUNDEX / JAROWINKLER_SIMILARITY
Fuzzy matching for dirty names and duplicate detection workflows.
SELECT name, JAROWINKLER_SIMILARITY(name, 'Jon Smyth') AS score FROM contacts;
06
AI_SUMMARIZE_AGG
The "TL;DR" for 2.4 Billion Rows.
SELECT AI_SUMMARIZE_AGG(review_text) FROM customer_reviews;
